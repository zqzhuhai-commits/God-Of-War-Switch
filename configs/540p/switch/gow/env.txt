# 战神 GOWS_169.41：中文配置。只修改等号右侧的值。
# 备注必须单独放在以 # 开头的行，不要写在数值后面。
# 改配置后退出游戏再启动；插拔底座后也需重启以应用倍率。
# 540p内部渲染：448行基准约540行；手持与底座输出均1280x720。内部尺寸随原始场景变化。

# 启动 ELF 的完整路径；必须与配套新资源匹配，不要改成其他版本的 ELF。
GOW_BOOT_ELF=sdmc:/switch/gow/disc-gows16941/GOWS_169.41
# 现有存档卡路径；沿用 play.mcd，不需要建立独立存档。
GOW_MC0=sdmc:/switch/gow/play.mcd

# 游戏设置：1=启动时默认开启 16:9 宽屏；0=不覆盖游戏原来的选项。
# 这是游戏的宽屏选项和相机比例，不只是把 4:3 图像拉宽。
GOW_WIDESCREEN=1
# 游戏设置：1=启动时默认开启 480p 逐行扫描；0=不覆盖游戏原来的选项。
# 由游戏自身切换视频预设；不会把 Switch 的 720p/1080p 输出降成 480p。
GOW_PROGRESSIVE=1

# 手持内部渲染倍率，可填 1.0 至 4.0 的数字，允许小数。
# 示例：1=原始尺寸，1.205357143≈540行，1.607142857≈720行，2=两倍（448行基准）。
# 2.410714286=按 448 行基准约 1080 行，3=三倍，4=四倍。
# 倍率不是固定宽高；实际内部尺寸随原始场景/帧缓冲变化。越高越吃性能和内存。
GOW_RENDER_SCALE_HANDHELD=1.205357143
# 底座内部倍率：同样支持 1.0 至 4.0，常用值同上。
# 本配置底座内部倍率=1.205357143，与手持一致。
GOW_RENDER_SCALE_DOCKED=1.205357143
# 手持输出帧缓冲高度，只能填 540 或 720；宽度自动按 16:9 计算。
# 540=960x540，720=1280x720；和内部倍率分别配置，不会自动互相修改。
GOW_OUTPUT_HEIGHT_HANDHELD=720
# 底座输出帧缓冲高度，只能填 540、720、1080。
# 对应960x540、1280x720、1920x1080；本版底座默认720。
GOW_OUTPUT_HEIGHT_DOCKED=720
# 最终放大滤镜：bilinear=双线性，开销较低；bicubic=双三次加锐化，开销较高。
GOW_SCALER=bilinear

# CPU 频率，单位 MHz：0=不改外部设置；max=程序预设 1785；也可填正数。
# 0 会保留你在超频工具中的设置；max 不是超频工具允许的最高值。
GOW_CPU_MHZ=0
# GPU 频率，单位 MHz：0=不改外部设置；max=手持 460.8/底座 768；也可填正数。
GOW_GPU_MHZ=0
# 内存控制器频率，单位 MHz：0=不改外部设置；max=1600；也可填正数。
GOW_EMC_MHZ=0
# CPU 调度：auto=允许线程使用系统授权的所有核心；pinned=按线程分配核心。
# auto 不会解锁系统保留核心；本机此前授权核心掩码为 0x7（核心 0、1、2）。
GOW_CPU_AFFINITY=auto

# 音频回调优先级提升档数，可填 0至9；0=不提升，本版2=比默认高两档。
# 自动调度仍允许使用所有授权核心；系统拒绝时会在日志记录错误。
GOW_PRIO_AudioOut=2
# IOP线程优先级提升档数，可填0至9；本版1=比默认高一档，优先供应设备/声音逻辑。
GOW_PRIO_IopRun=1
# CG 音轨号，0 至 255 的整数；本资源已由你试听确认 0=标准普通话。
# 不清楚其他轨道的语言映射时保持 0；只选一条音轨，仍保留其左右声道。
GOW_MPEG_AUDIO_TRACK=0
# 音频回调帧数，只支持 512、1024、2048、4096。
# 48kHz 下分别约 10.7、21.3、42.7、85.3ms；越大越能容忍短时抖动，但延迟增加。
# 加大缓冲不能解决持续供给不足。当前保留 2048 的测试设置。
GOW_AUDIO_BUFFER_FRAMES=2048
# IOP 执行：1=独立线程，0=与 EE 线程一起执行；IOP 负责声音和设备逻辑。
GOW_IOP_THREAD=1
# VU 顶点快速内核：1=启用，0=关闭作画面正确性对照，可能明显降低帧率。
GOW_VU1_HLE=1

# 帧率统计：frames=只记帧数，1=完整耗时诊断，0=关闭。
# 日常测试保持 frames；完整诊断有额外开销。
GOW_PERF_DIAG=frames
# 游戏翻帧函数地址；本版本固定使用 0x183BB8，请不要修改。
GOW_PERF_FRAME_PC=0x183BB8
# 游戏时间步诊断：0=关闭，1=开启；一般保持 0。
GOW_DELTA_DIAG=0
# 音频缺失/丢弃统计：0=关闭，1=每约 5 秒汇总，便于排查卡顿。
GOW_AUDIO_STATS=1
# 详细图形统计：0=关闭，1=开启；一般保持 0，避免额外开销。
GOW_GS_HW_STATS=0
# IOP 每批次耗时统计：0=关闭，1=开启；一般保持 0。
GOW_IOP_THREAD_STATS=0
# 主机 PC 位置采样频率（Hz）：0=关闭；诊断可用 20，开启会暂停线程采样。
GOW_PC_SAMPLE=0
# EE 游戏函数采样频率（Hz）：0=关闭；诊断可用 20；一般保持 0。
GOW_EE_SAMPLE=0
# 自动按键测试：0=关闭，正常游玩必须用 0；1=开发用自动按键。
GOW_PAD_TEST=0




# 显示同步间隔：1=原显示同步策略；2=60Hz下最高30次/秒。锁30未改善缺音，本版用1。
# 不丢弃客体显示请求，已撤回上一版GOW_PRESENT_MAX_HZ限频代码。
GOW_SWAP_INTERVAL=1
# 混音工作线程：恢复原设置1。
GOW_SOUND_THREAD=1


# 游戏帧时钟墙钟追赶：1=启用，0=关闭以对比旧时序；单帧上限约3.75个场周期。
# 渲染耗时时推进EE计时器与IOP音频；不是锁帧或插帧。
GOW_FRAME_AUDIO_CATCHUP=1

# 全屏调制分块边界对齐：1=启用竖线修正；0=关闭作对照。仅影响匹配的全屏批次。
GOW_POSTPROCESS_SNAP=1
